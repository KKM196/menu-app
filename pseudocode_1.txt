START
    SYSTEM = REGISTRATIONSYSTEM()
    SYSTEM.RUN()
END
CONTROL MODULE (RUN)
START
    CHOICE = 0
    WHILE CHOICE != 7
        PROCESS DISPLAY_MENU
        ENTER CHOICE
        IF CHOICE >= 1 AND CHOICE <= 7 THE
 CASE CHOICE OF
                1: PROCESS REGISTER_RUNNER
                2: PROCESS SEARCH_RUNNER
                3: PROCESS UPDATE_RUNNER
                4: PROCESS DELETE_RUNNER
                5: PROCESS DISPLAY_ALL_RUNNERS
                6: PROCESS GENERATE_REPORTS
                7: CONTINUE
       OTHERWISE: PRINT "INVALID CHOICE"
      ENDCASE
        ELSE
            PRINT "INVALID CHOICE. ENTER 1-7"
        ENDIF
    ENDWHILE
    PRINT "EXITING SYSTEM..."
END
DISPLAYMENU
START
    PRINT "=============================================="
     PRINT "   COMMUNITY RUNNING EVENT REGISTRATION SYSTEM"
    PRINT "================================================"
    PRINT " "
     PRINT "   1. REGISTER A RUNNER"
     PRINT "   2. SEARCH FOR A RUNNER"
      PRINT "   3. UPDATE RUNNER INFORMATION"
      PRINT "   4. DELETE A RUNNER"
     PRINT "   5. DISPLAY ALL RUNNERS"
     PRINT "   6. GENERATE REPORTS"
    PRINT "   7. EXIT SYSTEM"
    PRINT "  "
    PRINT "=============================================="
END
REGISTERRUNNER
START
    ENTER RUNNERNAME
    ENTER AGE
    ENTER GENDER
    ENTER CONTACTNUMBER
    ENTER EMAILADDRESS
    ENTER EMERGENCYCONTACT
    ENTER RACEDISTANCE
    ENTER CLUBMEMBER
    REGISTRATIONDATE = DATE()
    RACENUMBER = GENERATEUNIQUENUMBER()
    PROCESS VALIDATEAGEELIGIBILITY (AGE, RACEDISTANCE)
    IF VALIDAGE = TRUE THEN
        PROCESS DETERMINEAGECATEGORY (AGE)
        PROCESS DETERMINEBASEFEE (RACEDISTANCE)
        PROCESS DETERMINEHIGHESTDISCOUNT (AGE, CLUBMEMBER, REGISTRATIONDATE)
        DISCOUNTAMOUNT = BASEFEE * HIGHESTDISCOUNT
        FINALFEE = BASEFEE - DISCOUNTAMOUNT
        PROCESS STORERUNNERRECORD
        PRINT "REGISTRATION SUCCESSFUL!"
        PRINT "RACE NUMBER: ", RACENUMBER
        PRINT "AGE CATEGORY: ", AGECATEGORY
        PRINT "BASE FEE: R", BASEFEE
        PRINT "DISCOUNT APPLIED: ", HIGHESTDISCOUNT * 100, "%"
        PRINT "FINAL FEE: R", FINALFEE
    ELSE
        PRINT "REGISTRATION FAILED. AGE NOT ELIGIBLE."
    ENDIF
END

SEARCHRUNNER
START
    IF TOTALRUNNERS = 0 THEN
        PRINT "NO RUNNERS REGISTERED."
        RETURN
    ENDIF
    PRINT "SEARCH BY:"
    PRINT "   1. RACE NUMBER"
    PRINT "   2. RUNNER NAME"
    ENTER SEARCHTYPE
    IF SEARCHTYPE = 1 THEN
        ENTER SEARCHNUMBER
        PROCESS FINDBYNUMBER (SEARCHNUMBER)
    ELSE
        ENTER SEARCHNAME
        PROCESS FINDBYNAME (SEARCHNAME)
    ENDIF
    IF FOUND = TRUE THEN
        PROCESS DISPLAYRUNNERDETAILS (INDEX)
    ELSE
        PRINT "RUNNER NOT FOUND.
 ENDIF
END
UPDATERUNNER
START
    IF TOTALRUNNERS = 0 THEN
        PRINT "NO RUNNERS REGISTERED."
        RETURN
    ENDIF
    ENTER SEARCHNUMBER
    PROCESS FINDBYNUMBER (SEARCHNUMBER)
    IF FOUND = TRUE THEN
        PROCESS DISPLAYRUNNERDETAILS (INDEX)
        ENTER NEWRUNNERNAME (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWAGE (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWGENDER (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWCONTACTNUMBER (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWEMAILADDRESS (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWEMERGENCYCONTACT (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWRACEDISTANCE (LEAVE BLANK TO KEEP CURRENT)
        ENTER NEWCLUBMEMBER (LEAVE BLANK TO KEEP CURRENT)
        PROCESS UPDATERECORD (INDEX)
        PRINT "UPDATE SUCCESSFUL!"
    ELSE
        PRINT "RUNNER NOT FOUND."
    ENDIF
END
DELETERUNNER
START
 IF TOTALRUNNERS = 0 THEN
        PRINT "NO RUNNERS REGISTERED."
        RETURN
    ENDIF
    ENTER SEARCHNUMBER
    PROCESS FINDBYNUMBER (SEARCHNUMBER)
    IF FOUND = TRUE THEN
        PROCESS DISPLAYRUNNERDETAILS (INDEX)
        PRINT "ARE YOU SURE YOU WANT TO DELETE THIS RUNNER? (Y/N)"
        ENTER CONFIRMATION
        IF CONFIRMATION = "Y" THEN
            PROCESS DELETERECORD (INDEX)
            PRINT "RUNNER DELETED."
        ELSE
            PRINT "DELETION CANCELLED."
        ENDIF
    ELSE
        PRINT "RUNNER NOT FOUND."
    ENDIF
END
DISPLAYALLRUNNERS
START
    IF TOTALRUNNERS = 0 THEN
        PRINT "NO RUNNERS REGISTERED."
        RETURN
    ENDIF
    PRINT HEADINGS
    PRINT "RACE #", "NAME", "AGE", "DISTANCE", "FEE"
    FOR RUNNER = 1 TO TOTALRUNNERS
        PRINT RUNNERDETAILS (RUNNER)
    ENDFOR
    PRINT "TOTAL RUNNERS: ", TOTALRUNNERS
    PRINT "TOTAL INCOME: R", TOTALINCOME
END

GENERATEREPORTS
START
    IF TOTALRUNNERS = 0 THEN
        PRINT "NO RUNNERS REGISTERED."
        RETURN
    ENDIF
    PRINT "   1. RUNNER LIST REPORT"
    PRINT "   2. AGE CATEGORY REPORT"
    PRINT "   3. DISCOUNT REPORT"
    PRINT "   4. INCOME REPORT"
    PRINT "   5. RETURN TO MAIN MENU"
    ENTER REPORTCHOICE
    CASE REPORTCHOICE OF
        1: PROCESS RUNNERLISTREPORT
        2: PROCESS AGECATEGORYREPORT
        3: PROCESS DISCOUNTREPORT
        4: PROCESS INCOMEREPORT
        5: CONTINUE
        OTHERWISE: PRINT "INVALID CHOICE"
    ENDCASE
END

FINDBYNUMBER (SEARCHNUMBER, *FOUND, *INDEX)
START
    FOUND = FALSE
    RUNNER = 1
    WHILE RUNNER <= TOTALRUNNERS AND FOUND = FALSE
        IF RUNNER(RUNNER).RACENUMBER = SEARCHNUMBER THEN
            FOUND = TRUE
            INDEX = RUNNER
        ELSE
            RUNNER = RUNNER + 1
        ENDIF
    ENDWHILE
    RETURN FOUND, INDEX
END

FINDBYNAME (SEARCHNAME, *FOUND, *INDEX)
START
    FOUND = FALSE
    RUNNER = 1
    WHILE RUNNER <= TOTALRUNNERS AND FOUND = FALSE
        IF RUNNER(RUNNER).RUNNERNAME = SEARCHNAME THEN
            FOUND = TRUE
            INDEX = RUNNER
        ELSE
            RUNNER = RUNNER + 1
        ENDIF
    ENDWHILE
    RETURN FOUND, INDEX
END

VALIDATEAGEELIGIBILITY (AGE, RACEDISTANCE, *VALIDAGE)
START
    VALIDAGE = FALSE
    IF RACEDISTANCE = 5 THEN
        IF AGE >= 8 THEN
            VALIDAGE = TRUE
        ENDIF
    ELSE
        IF RACEDISTANCE = 10 THEN
            IF AGE >= 14 THEN
                VALIDAGE = TRUE
            ENDIF
        ELSE
            IF RACEDISTANCE = 21 THEN
                IF AGE >= 16 THEN
                    VALIDAGE = TRUE
                ENDIF
            ELSE
                IF RACEDISTANCE = 42 THEN
                    IF AGE >= 20 THEN
                        VALIDAGE = TRUE
                    ENDIF
                ENDIF
            ENDIF
        ENDIF
    ENDIF
    IF VALIDAGE = FALSE THEN
        PRINT "AGE NOT ELIGIBLE FOR SELECTED RACE DISTANCE."
    ENDIF
    RETURN VALIDAGE
END

DETERMINEAGECATEGORY (AGE, *AGECATEGORY)
START
    IF AGE >= 60 THEN
        AGECATEGORY = "SENIOR"
    ELSE
        IF AGE >= 40 THEN
            AGECATEGORY = "VETERAN"
        ELSE
            IF AGE >= 18 THEN
                AGECATEGORY = "OPEN"
            ELSE
                AGECATEGORY = "JUNIOR"
            ENDIF
        ENDIF
    ENDIF
    RETURN AGECATEGORY
END

DETERMINEBASEFEE (RACEDISTANCE, *BASEFEE)
START
    IF RACEDISTANCE = 5 THEN
        BASEFEE = BASEFEE_5KM
    ELSE
        IF RACEDISTANCE = 10 THEN
            BASEFEE = BASEFEE_10KM
        ELSE
            IF RACEDISTANCE = 21 THEN
                BASEFEE = BASEFEE_HALFMARATHON
            ELSE
                IF RACEDISTANCE = 42 THEN
                    BASEFEE = BASEFEE_MARATHON
                ELSE
                    BASEFEE = 0
                ENDIF
            ENDIF
        ENDIF
    ENDIF
    RETURN BASEFEE
END

DETERMINEHIGHESTDISCOUNT (AGE, CLUBMEMBER, REGISTRATIONDATE, *HIGHESTDISCOUNT)
START
    HIGHESTDISCOUNT = 0
    IF AGE >= 60 THEN
        HIGHESTDISCOUNT = DISCOUNT_SENIOR
    ELSE
        IF AGE < 18 THEN
            HIGHESTDISCOUNT = DISCOUNT_UNDER18
        ENDIF
    ENDIF
    IF CLUBMEMBER = TRUE THEN
        IF HIGHESTDISCOUNT < DISCOUNT_CLUBMEMBER THEN
            HIGHESTDISCOUNT = DISCOUNT_CLUBMEMBER
        ENDIF
    ENDIF
    IF REGISTRATIONDATE <= EVENT_DATE - 30 THEN
        IF HIGHESTDISCOUNT < DISCOUNT_EARLYBIRD THEN
            HIGHESTDISCOUNT = DISCOUNT_EARLYBIRD
        ENDIF
    ENDIF
    RETURN HIGHESTDISCOUNT
END
CALCULATEFINALFEE (RACEDISTANCE, AGE, CLUBMEMBER, REGISTRATIONDATE, *FINALFEE)
START
    PROCESS DETERMINEBASEFEE (RACEDISTANCE, *BASEFEE)
    PROCESS DETERMINEHIGHESTDISCOUNT (AGE, CLUBMEMBER, REGISTRATIONDATE, *HIGHESTDISCOUNT)
    DISCOUNTAMOUNT = BASEFEE * HIGHESTDISCOUNT
    FINALFEE = BASEFEE - DISCOUNTAMOUNT
    RETURN FINALFEE
END

START
    PRINT "RACE NUMBER: ", RUNNER(RACENUMBER)
    PRINT "NAME: ", RUNNER(RUNNERNAME)
    PRINT "AGE: ", RUNNER(AGE)
    PRINT "GENDER: ", RUNNER(GENDER)
    PRINT "CONTACT: ", RUNNER(CONTACTNUMBER)
    PRINT "EMAIL: ", RUNNER(EMAILADDRESS)
    PRINT "EMERGENCY: ", RUNNER(EMERGENCYCONTACT)
    PRINT "RACE DISTANCE: ", RUNNER(RACEDISTANCE), " KM"
    PRINT "CLUB MEMBER: ", RUNNER(CLUBMEMBER)
    PRINT "AGE CATEGORY: ", RUNNER(AGECATEGORY)
    PRINT "BASE FEE: R", RUNNER(BASEFEE)
    PRINT "DISCOUNT: ", RUNNER(HIGHESTDISCOUNT) * 100, "%"
    PRINT "DISCOUNT AMOUNT: R", RUNNER(DISCOUNTAMOUNT)
    PRINT "FINAL FEE: R", RUNNER(FINALFEE)
END

START

    FOR RUNNER = 1 TO TOTALRUNNERS
        PRINT RUNNERDETAILS (RUNNER)
    ENDFOR

    PRINT "TOTAL INCOME: R", TOTALINCOME
END
AGECATEGORYREPORT
START
    JUNIOR = 0
    OPEN = 0
    VETERAN = 0
    SENIOR = 0
    FOR RUNNER = 1 TO TOTALRUNNERS
        IF RUNNER(AGECATEGORY) = "JUNIOR" THEN
            JUNIOR = JUNIOR + 1
        ELSE
            IF RUNNER(AGECATEGORY) = "OPEN" THEN
                OPEN = OPEN + 1
            ELSE
                IF RUNNER(AGECATEGORY) = "VETERAN" THEN
                    VETERAN = VETERAN + 1
                ELSE
                    SENIOR = SENIOR + 1
                ENDIF
            ENDIF
        ENDIF
    ENDFOR
    PRINT "AGE CATEGORY REPORT"
    PRINT "JUNIOR: ", JUNIOR
    PRINT "OPEN: ", OPEN
    PRINT "VETERAN: ", VETERAN
    PRINT "SENIOR: ", SENIOR
END

INCOMEREPORT
START
    TOTAL5KM = 0
    TOTAL10KM = 0
    TOTALHALF = 0
    TOTALMARATHON = 0
    FOR RUNNER = 1 TO TOTALRUNNERS
        IF RUNNER(RACEDISTANCE) = 5 THEN
            TOTAL5KM = TOTAL5KM + RUNNER(FINALFEE)
        ELSE
            IF RUNNER(RACEDISTANCE) = 10 THEN
                TOTAL10KM = TOTAL10KM + RUNNER(FINALFEE)
            ELSE
                IF RUNNER(RACEDISTANCE) = 21 THEN
                    TOTALHALF = TOTALHALF + RUNNER(FINALFEE)
                ELSE
                    TOTALMARATHON = TOTALMARATHON + RUNNER(FINALFEE)
                ENDIF
            ENDIF
        ENDIF
    ENDFOR
    PRINT "INCOME REPORT"
    PRINT "5 KM: R", TOTAL5KM
    PRINT "10 KM: R", TOTAL10KM
    PRINT "HALF MARATHON: R", TOTALHALF
    PRINT "MARATHON: R", TOTALMARATHON
    PRINT "TOTAL INCOME: R", TOTALINCOME
END

DISCOUNT_CLUBMEMBER = 0.10
DISCOUNT_SENIOR = 0.20
DISCOUNT_EARLYBIRD = 0.10

MINIMUM_AGE_5KM = 8
MINIMUM_AGE_10KM = 14
MINIMUM_AGE_HALFMARATHON = 16
MINIMUM_AGE_MARATHON = 20

EVENT_DATE = "2026-08-30"
EARLYBIRD_DAYS = 30
